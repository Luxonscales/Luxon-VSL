    var preview = window.LX_PREVIEW === true || /[?&]lx-preview=1(&|$)/.test(window.location.search);

    /* 1. Every CTA goes to one URL, and carries the ad's tracking with it.
          Only these keys are passed on, so nothing else in the address bar
          leaks into the application. */
    var KEYS = ['utm_source', 'utm_medium', 'utm_campaign', 'utm_content', 'utm_term', 'fbclid'];
    var applyUrl = root.getAttribute('data-lx-apply') || '/apply';
    function ctaHref() {
      var here = new URLSearchParams(window.location.search), u;
      try { u = new URL(applyUrl, window.location.href); } catch (e) { return applyUrl; }
      KEYS.forEach(function (k) {
        var v = here.get(k);
        if (v !== null && v !== '' && !u.searchParams.has(k)) u.searchParams.set(k, v);
      });
      return u.origin === window.location.origin ? u.pathname + u.search + u.hash : u.toString();
    }
    function wireCtas(scope) {
      var href = ctaHref();
      each(scope.querySelectorAll('a[data-lx-cta]'), function (a) { a.setAttribute('href', href); });
    }

    each(root.querySelectorAll('img[data-lx-src]'), function (img) {

      var slot = img.closest('[data-lx-slot]') || img.parentNode;
      if (isPlaceholder(src)) {

        slot.setAttribute('data-lx-need', src.replace(/^\s*\[|\]\s*$/g, ''));
        if (slot.tagName === 'BUTTON') slot.disabled = true;
        return;

      if (box.querySelector('video,iframe')) return;

        return;

      btn.type = 'button';
      btn.className = 'lx-lp-embed__play';
      btn.setAttribute('aria-label', 'Play the ' + label.toLowerCase());

        p.className = 'lx-lp-embed__poster';
        p.alt = '';
        p.decoding = 'async';

        btn.appendChild(p);
      }

    var lastFocus = null;
    if (dlg) {
      dlg.querySelector('.lx-lp-lightbox__close').addEventListener('click', function () { dlg.close(); });
      dlg.addEventListener('click', function (ev) { if (ev.target === dlg) dlg.close(); });
      dlg.addEventListener('close', function () { dimg.removeAttribute('src'); if (lastFocus) lastFocus.focus(); });
    }
    each(root.querySelectorAll('[data-lx-zoom]'), function (b) {
      b.addEventListener('click', function () {
        var img = b.querySelector('img');
        var src = img && (img.currentSrc || img.getAttribute('src'));
        if (!src) return;
        if (!dlg || typeof dlg.showModal !== 'function') { window.open(src, '_blank', 'noopener'); return; }
        lastFocus = b;
        dimg.src = src; dimg.alt = img.alt;
        dlg.showModal();
      });
    });

    /* 5. Reveal on scroll. Skipped entirely for reduced motion. */
    var items = root.querySelectorAll('[data-lx-reveal]');
    if (reduce || !('IntersectionObserver' in window)) {
      each(items, function (el) { el.classList.add('is-in'); });
    } else {
      var io = new IntersectionObserver(function (entries) {
        entries.forEach(function (en) {
          if (en.isIntersecting) { en.target.classList.add('is-in'); io.unobserve(en.target); }
        });
      }, { rootMargin: '0px 0px -8% 0px', threshold: 0.06 });
      each(items, function (el) { io.observe(el); });
    }

    /* 6. Sticky bar: moved to <body> so position:fixed is never trapped by a
          transformed GHL container. Up once the hero has left the screen,
          down again from the final CTA onward, where the real button is. */
    var sticky = root.querySelector('[data-lx-sticky]');
    if (sticky) {
      var old = document.getElementById('lx-lp-sticky');
      if (old && old !== sticky && old.parentNode) old.parentNode.removeChild(old);
      sticky.id = 'lx-lp-sticky';
      document.body.appendChild(sticky);
      wireCtas(sticky);
      var heroGone = false, finalReached = false;
      var sync = function () {
        var on = heroGone && !finalReached;
        sticky.classList.toggle('is-up', on);
        root.classList.toggle('has-sticky', on);
        if (on) { sticky.removeAttribute('inert'); sticky.setAttribute('aria-hidden', 'false'); }
        else { sticky.setAttribute('inert', ''); sticky.setAttribute('aria-hidden', 'true'); }
      };
      var hero = root.querySelector('[data-lx-hero]');
      var fin = root.querySelector('[data-lx-final]');
      if ('IntersectionObserver' in window && hero) {
        new IntersectionObserver(function (e) { heroGone = !e[0].isIntersecting && e[0].boundingClientRect.top < 0; sync(); }).observe(hero);
        if (fin) new IntersectionObserver(function (e) {
          finalReached = e[0].isIntersecting || e[0].boundingClientRect.top < 0; sync();
        }).observe(fin);
      } else { heroGone = true; sync(); }
    }

    /* Optional bits (the contact email) disappear on the live page while
       they still hold a placeholder, instead of printing brackets. */
    each(root.querySelectorAll('[data-lx-optional]'), function (el) {
      if (!preview && /\[/.test(el.textContent + (el.getAttribute('href') || ''))) el.hidden = true;
    });

    /* 7. Footer year. */
    each(root.querySelectorAll('[data-lx-year]'), function (el) { el.textContent = new Date().getFullYear(); });

    /* 8. Anything still in [BRACKETS] gets listed in the console, so nothing
          ships unfinished by accident. */
    var left = [];
    var walk = document.createTreeWalker(root, NodeFilter.SHOW_TEXT);
    while (walk.nextNode()) { var t = walk.currentNode.nodeValue.match(/\[[^\]]{2,}\]/g); if (t) left = left.concat(t); }
    each(root.querySelectorAll('[data-lx-src],[data-lx-embed],[data-lx-poster],a[href^="mailto:["]'), function (el) {
      ['data-lx-src', 'data-lx-embed', 'data-lx-poster', 'href'].forEach(function (a) {
        var v = el.getAttribute(a); if (v && /\[/.test(v)) left.push(v);
      });
    });
    if (left.length && window.console && console.warn) {
      console.warn('[Luxon landing] ' + left.length + ' placeholder(s) still on this page:\n  ' +
        left.filter(function (v, i) { return left.indexOf(v) === i; }).join('\n  '));
    }
  }

  if (document.readyState === 'loading') document.addEventListener('DOMContentLoaded', init);
  else init();
})();
</script>
